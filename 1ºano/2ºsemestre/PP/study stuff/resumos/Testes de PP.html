public class StrategyImpl implements Strategy {

    private boolean hasCollectableContainer(Vehicle vehicle, AidBox aidbox) {
        Container[] containers = aidbox.getContainers();
        if (containers == null) return false;

        for (int i = 0; i < containers.length; i++) {
            if (containers[i] != null && containers[i].getType() == vehicle.getSupplyType()) {
                Measurement measurement = containers[i].getLastMeasurement();
                if (measurement != null) {
                    double valor = measurement.getValue();
                    double limiteMaximo = containers[i].getCapacity() * 0.8;
                    if (valor > limiteMaximo) {
                        return true;
                    }
                }
            }
        }
        return false;
    }

    private boolean addAidBoxToRoute(Route route, AidBox aidbox, RouteValidator validator) {
        if (validator == null || route == null || aidbox == null) return false;
        
        if (validator.validate(route, aidbox)) {
            try {
                route.addAidBox(aidbox);
                return true;
            } catch (RouteException e) {
                return false;
            }
        }
        return false;
    }
}

O encapsulamento garante a ocultação do estado interno de um objeto através do uso de modificadores de acesso restritivos como o private para proteger os dados contra modificações arbitrárias externas, permitindo o acesso direto apenas à própria classe, enquanto o modificador protected estende essa visibilidade para subclasses e classes do mesmo pacote, contrastando com o public que expõe o elemento de forma irrestrita a qualquer componente da aplicação.

Desenvolva na classe administrativa da instituição um método público chamado calculateTotalCapacityUsed. Este método recebe como argumento um array convencional de caixas de suprimentos (AidBox[]) e um determinado ItemType. O método deve percorrer todas as estruturas e retornar a soma total acumulada (double) dos valores de capacidade de todos os contentores cujo tipo coincida com o solicitado. Garanta o tratamento de posições nulas. 
public double calculateTotalCapacityUsed(AidBox[] boxes, ItemType filterType) {
    if (boxes == null || filterType == null) return 0.0;
    double totalCapacity = 0.0;

    for (int i = 0; i < boxes.length; i++) {
        if (boxes[i] == null) continue;
        Container[] containers = boxes[i].getContainers();
        if (containers == null) continue;

        for (int j = 0; j < containers.length; j++) {
            if (containers[j] != null && containers[j].getType() == filterType) {
                totalCapacity += containers[j].getCapacity();
            }
        }
    }
    return totalCapacity;
}

public RefrigeratedVehicle[] extractActiveRefrigeratedVehicles(Vehicle[] vehicles) {
    if (vehicles == null) return new RefrigeratedVehicle[0];
    
    // Primeira passagem: Contagem exata para dimensionar o array nativo
    int count = 0;
    for (int i = 0; i < vehicles.length; i++) {
        if (vehicles[i] instanceof RefrigeratedVehicle) {
            RefrigeratedVehicleImpl refVeh = (RefrigeratedVehicleImpl) vehicles[i];
            if (refVeh.getState() == VehicleState.ENABLED) {
                count++;
            }
        }
    }

    // Alocação com o tamanho preciso
    RefrigeratedVehicle[] result = new RefrigeratedVehicle[count];
    int index = 0;

    // Segunda passagem: Preenchimento do vetor
    for (int i = 0; i < vehicles.length; i++) {
