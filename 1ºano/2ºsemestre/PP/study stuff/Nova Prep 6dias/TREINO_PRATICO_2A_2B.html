🚀 O Algoritmo Universal em 4 Passos (Resolve QUALQUER 2a e 2b)

Passo 1: A Guarda de Segurança (Null Checks)
Verifique sempre se os parâmetros recebidos (ou os arrays dos seus getters) valem null. Se forem nulos, retorne false, 0 ou new Tipo[0].
Passo 2: Na 2a — Métodos Auxiliares de Decisão
Crie funções pequenas que devolvem boolean (ex: esta caixa precisa de recolha?) ou valores (ex: calcular ocupação média). Percorra o array interno ignorando nulls.
Passo 3: Na 2b — Loop com Array Temporário
Crie um array temp com a capacidade máxima (ex: inst.getAidBoxes().length) e um contador int count = 0;. Guarde os elementos válidos com temp[count++].
Passo 4: Compactação do Array Final
Aloque o array final com o tamanho exato do contador (new Tipo[count]), copie os elementos de temp e retorne. Posições nulas eliminadas!

Devolve true se existir a ocorrência de, pelo menos, um contentor cujo tipo seja igual ao do veículo (getSupplyType()) e se a sua última medição registada tiver um valor superior a 80% da sua capacidade. Devolve false caso contrário ou se nulo.

public class StrategyImpl implements Strategy {

    public boolean hasCollectableContainer(Vehicle vehicle, AidBox aidbox) {
        if (vehicle == null || aidbox == null || aidbox.getContainers() == null) return false;
        
        Container[] containers = aidbox.getContainers();
        for (int i = 0; i < containers.length; i++) {
            Container c = containers[i];
            if (c != null && c.getType() == vehicle.getSupplyType()) {
                Measurement last = c.getLastMeasurement();
                if (last != null && last.getValue() > (c.getCapacity() * 0.8)) {
                    return true;
                }
            }
        }
        return false;
    }

    public boolean addAidBoxToRoute(Route route, AidBox aidbox, RouteValidator validator) {
        if (route == null || aidbox == null || validator == null) return false;
        if (!validator.validate(route, aidbox)) return false;
        try {
            route.addAidBox(aidbox);
            return true;
        } catch (RouteException e) {
            return false;
        }
    }

    @Override
    public Route[] generate(IInstitution inst, RouteValidator validator) {
        if (inst == null || validator == null) return new Route[0];
        Vehicle[] vehicles = inst.getVehicles();
        AidBox[] aidBoxes = inst.getAidBoxes();
        if (vehicles == null || aidBoxes == null) return new Route[0];

        Route[] temp = new Route[vehicles.length];
        int count = 0;

        for (int i = 0; i < vehicles.length; i++) {
            Vehicle v = vehicles[i];
            if (v == null) continue;
            
            Route currentRoute = new RouteImpl(v);
            for (int j = 0; j < aidBoxes.length; j++) {
                AidBox box = aidBoxes[j];
                if (box != null && hasCollectableContainer(v, box)) {
                    addAidBoxToRoute(currentRoute, box, validator);
                }
            }
            if (currentRoute.getRoute() != null && currentRoute.getRoute().length > 0) {
                temp[count++] = currentRoute;
            }
        }

        Route[] result = new Route[count];
        for (int i = 0; i < count; i++) result[i] = temp[i];
        return result;
    }
}

public class ReportImpl implements Report {

    public int countContainersByType(AidBox box, ItemType type) {
        if (box == null || type == null || box.getContainers() == null) return 0;
        int count = 0;
        for (Container c : box.getContainers()) {
            if (c != null && c.getType() == type) count++;
        }
        return count;
    }

    public double getAverageOccupancy(AidBox box) {
        if (box == null || box.getContainers() == null) return 0.0;
        double somaPercentagens = 0;
        int validos = 0;

        for (Container c : box.getContainers()) {
            if (c != null && c.getCapacity() > 0) {
                Measurement last = c.getLastMeasurement();
                if (last != null) {
                    somaPercentagens += (last.getValue() / c.getCapacity()) * 100.0;
                    validos++;
                }
            }
        }
        return validos == 0 ? 0.0 : somaPercentagens / validos;
    }

    @Override
    public String generate(IInstitution inst) {
        if (inst == null || inst.getAidBoxes() == null) return "";
        
        StringBuilder sb = new StringBuilder();
        AidBox[] boxes = inst.getAidBoxes();

        for (int i = 0; i < boxes.length; i++) {
            AidBox box = boxes[i];
            if (box != null && getAverageOccupancy(box) > 50.0) {
                sb.append("AidBox Code: ").append(box.getCode())
                  .append(" | Zona: ").append(box.getZone())
                  .append(" | Ocupação Média: ").append(getAverageOccupancy(box)).append("%\n");
            }
        }
        return sb.toString();
    }
}

public class CollectionManagerImpl implements CollectionManager {

    private double getContainerLoad(Container c) {
        if (c == null) return 0.0;
        Measurement last = c.getLastMeasurement();
        return last == null ? 0.0 : last.getValue();
    }

    private boolean isContainerFull(Container c, double threshold) {
        if (c == null || c.getCapacity() <= 0) return false;
        Measurement last = c.getLastMeasurement();
        if (last == null) return false;
        return last.getValue() > (c.getCapacity() * threshold / 100.0);
    }

    @Override
    public double getTotalCollectedByType(IInstitution inst, ItemType type) {
        if (inst == null || type == null || inst.getAidBoxes() == null) return 0.0;
        
        double totalCarga = 0.0;
        AidBox[] boxes = inst.getAidBoxes();

        for (int i = 0; i < boxes.length; i++) {
            if (boxes[i] == null) continue;
            Container[] cs = boxes[i].getContainers();
            if (cs == null) continue;

            for (int j = 0; j < cs.length; j++) {
                Container c = cs[j];
                if (c != null && c.getType() == type && isContainerFull(c, 75.0)) {
                    totalCarga += getContainerLoad(c);
                }
            }
        }
        return totalCarga;
    }
}

public class PickingMapImpl implements PickingMap {

    private boolean hasContainerOfType(AidBox box, ItemType type) {
        if (box == null || type == null || box.getContainers() == null) return false;
        for (Container c : box.getContainers()) {
            if (c != null && c.getType() == type) return true;
        }
        return false;
    }

    private boolean needsCollection(Container c) {
        if (c == null) return false;
        Measurement last = c.getLastMeasurement();
        if (last == null) return false;
        if (c.getType() == ItemType.PERISHABLE_FOOD) return true;
        return last.getValue() > (c.getCapacity() * 0.7);
    }

    @Override
    public AidBox[] getPickingMap(IInstitution inst, ItemType type) {
        if (inst == null || type == null || inst.getAidBoxes() == null) return new AidBox[0];
        
        AidBox[] boxes = inst.getAidBoxes();
        AidBox[] temp = new AidBox[boxes.length];
        int count = 0;

        for (int i = 0; i < boxes.length; i++) {
            AidBox box = boxes[i];
            if (box == null || !hasContainerOfType(box, type)) continue;
            
            Container[] cs = box.getContainers();
            if (cs == null) continue;

            boolean necessitaRecolha = false;
            for (Container c : cs) {
                if (c != null && c.getType() == type && needsCollection(c)) {
                    necessitaRecolha = true;
                    break;
                }
            }
            if (necessitaRecolha) temp[count++] = box;
        }

        AidBox[] result = new AidBox[count];
        for (int i = 0; i < count; i++) result[i] = temp[i];
        return result;
    }
}

public class AlertManagerImpl implements AlertManager {

    public boolean isContainerInCriticalState(Container c) {
        if (c == null) return true;
        Measurement last = c.getLastMeasurement();
        if (last == null || c.getCapacity() <= 0) return true;
        return (last.getValue() / c.getCapacity()) * 100.0 > 95.0;
    }

    public int countCriticalContainersInAidBox(AidBox box) {
        if (box == null || box.getContainers() == null) return 0;
        int count = 0;
        for (Container c : box.getContainers()) {
            if (isContainerInCriticalState(c)) count++;
        }
        return count;
    }

    @Override
    public Alert[] generateMaintenanceAlerts(IInstitution inst) {
        if (inst == null || inst.getAidBoxes() == null) return new Alert[0];
        
        AidBox[] boxes = inst.getAidBoxes();
        int totalAlertas = 0;

        for (AidBox box : boxes) {
            if (box != null && countCriticalContainersInAidBox(box) > 0) totalAlertas++;
        }

        if (totalAlertas == 0) return new Alert[0];

        Alert[] result = new Alert[totalAlertas];
        int idx = 0;

        for (AidBox box : boxes) {
            if (box != null) {
                int numCriticos = countCriticalContainersInAidBox(box);
                if (numCriticos > 0) {
                    String cod = "ALT-" + box.getCode();
                    String desc = "AidBox em risco com " + numCriticos + " contentores criticos.";
                    int gravidade = Math.min(numCriticos, 5);
                    
                    result[idx++] = new AlertImpl(cod, AlertType.CAPACITY_OVERFLOW, desc, gravidade);
                }
            }
        }
        return result;
    }
}

public class EmergencyStrategyImpl implements EmergencyStrategy {

    public double getAidBoxEmergencyVolume(AidBox aidbox, ItemType type) {
        if (aidbox == null || type == null || aidbox.getContainers() == null) return 0.0;
        
        double totalVolume = 0.0;
        Container[] containers = aidbox.getContainers();

        for (int i = 0; i < containers.length; i++) {
            Container c = containers[i];
            if (c != null && c.getType() == type && c.getCapacity() > 0) {
                Measurement last = c.getLastMeasurement();
                if (last != null) {
                    double ocupacao = (last.getValue() / c.getCapacity()) * 100.0;
                    if (ocupacao > 85.0) {
                        totalVolume += last.getValue();
                    }
                }
            }
        }
        return totalVolume;
    }

    public boolean tryAssignAidBoxToRoute(Route route, AidBox aidbox, RouteValidator validator, double maxDistance) {
        if (route == null || aidbox == null || validator == null) return false;
        if (aidbox.getDistanceToBase() > maxDistance) return false;
        if (!validator.validate(route, aidbox)) return false;
        
        try {
            route.addAidBox(aidbox);
            return true;
        } catch (RouteException e) {
            return false;
        }
    }

    @Override
    public Route[] generateEmergencyRoutes(IInstitution inst, RouteValidator validator, double maxDistance) {
        if (inst == null || validator == null) return new Route[0];
        Vehicle[] vehicles = inst.getVehicles();
        AidBox[] aidBoxes = inst.getAidBoxes();
        if (vehicles == null || aidBoxes == null) return new Route[0];

        Route[] temp = new Route[vehicles.length];
        int count = 0;

        for (int i = 0; i < vehicles.length; i++) {
            Vehicle v = vehicles[i];
            if (v == null) continue;

            Route currentRoute = new RouteImpl(v);
            for (int j = 0; j < aidBoxes.length; j++) {
                AidBox box = aidBoxes[j];
                if (box != null) {
                    double emergVolume = getAidBoxEmergencyVolume(box, v.getSupplyType());
                    if (emergVolume > 0.0) {
                        tryAssignAidBoxToRoute(currentRoute, box, validator, maxDistance);
                    }
                }
            }

            if (currentRoute.getRoute() != null && currentRoute.getRoute().length > 0) {
                temp[count++] = currentRoute;
            }
        }

        Route[] result = new Route[count];
        for (int i = 0; i < count; i++) {
            result[i] = temp[i];
        }
        return result;
    }
}

public class AuditManagerImpl implements AuditManager {

    public boolean isSensorFaulty(Container container) {
        if (container == null || container.getCapacity() <= 0) return true;
        Measurement last = container.getLastMeasurement();
        if (last == null) return true;
        
        if (last.getSensorStatus() == SensorStatus.HARDWARE_FAULT) return true;
        
        double ocupacao = (last.getValue() / container.getCapacity()) * 100.0;
        return ocupacao > 100.0;
    }

    public double getAverageFaultOccupancyInAidBox(AidBox box) {
        if (box == null || box.getContainers() == null) return 0.0;
        
        double somaPercentagens = 0.0;
        int countFalhas = 0;
        Container[] containers = box.getContainers();

        for (int i = 0; i < containers.length; i++) {
            Container c = containers[i];
            if (isSensorFaulty(c)) {
                countFalhas++;
                if (c != null && c.getCapacity() > 0 && c.getLastMeasurement() != null) {
                    somaPercentagens += (c.getLastMeasurement().getValue() / c.getCapacity()) * 100.0;
                }
            }
        }
        return countFalhas == 0 ? 0.0 : somaPercentagens / countFalhas;

public class RegionalDistributionManagerImpl implements RegionalDistributionManager {

public class QualityAuditManagerImpl implements QualityAuditManager {

    public boolean isUnreliableSensor(Container container) {
        if (container == null || container.getCapacity() <= 0) return true;
        Measurement last = container.getLastMeasurement();
        if (last == null) return true;
        
        if (last.getSensorStatus() != SensorStatus.OK) return true;
        if (last.getValue() < 0.0 || last.getValue() > container.getCapacity()) return true;

        return false;
    }

    public double calculateUnreliablePercentage(AidBox box) {
        if (box == null || box.getContainers() == null || box.getContainers().length == 0) {
            return 0.0;
        }

        Container[] containers = box.getContainers();
        int total = containers.length;
        int unreliableCount = 0;

        for (int i = 0; i < total; i++) {
            if (isUnreliableSensor(containers[i])) {
                unreliableCount++;
            }
        }
        return ((double) unreliableCount / total) * 100.0;
    }

    @Override
    public QualityReport[] generateQualityAudit(IInstitution inst, double minUnreliableThreshold) {
        if (inst == null || inst.getAidBoxes() == null) {
            return new QualityReport[0];
        }

        AidBox[] boxes = inst.getAidBoxes();
        QualityReport[] temp = new QualityReport[boxes.length];
        int count = 0;

        for (int i = 0; i < boxes.length; i++) {
            AidBox box = boxes[i];
            if (box != null && box.getContainers() != null) {
                double percFalha = calculateUnreliablePercentage(box);
                if (percFalha > minUnreliableThreshold) {
                    int totalCs = box.getContainers().length;
                    int falhas = 0;
                    for (Container c : box.getContainers()) {
                        if (isUnreliableSensor(c)) falhas++;
                    }
                    temp[count++] = new QualityReportImpl(box.getCode(), totalCs, falhas, percFalha);
                }
            }
        }

        // Copiar para array final
        QualityReport[] result = new QualityReport[count];
        for (int i = 0; i < count; i++) {
            result[i] = temp[i];
        }

        // Ordenação Manual Decrescente por Percentagem de Falha (Bubble Sort sem java.util)
        for (int i = 0; i < count - 1; i++) {
            for (int j = 0; j < count - 1 - i; j++) {
                if (result[j].getUnreliablePercentage() < result[j + 1].getUnreliablePercentage()) {
                    QualityReport tempReport = result[j];
                    result[j] = result[j + 1];
                    result[j + 1] = tempReport;
                }
            }
        }

        return result;
    }
}

public class EmergencyDonationDispatcherImpl implements EmergencyDonationDispatcher {

    public double getAvailableCapacityForItemType(AidBox box, ItemType type) {
        if (box == null || type == null || box.getContainers() == null) return 0.0;

        double disponivelTotal = 0.0;
        Container[] containers = box.getContainers();

        for (int i = 0; i < containers.length; i++) {
            Container c = containers[i];
            if (c != null && c.getType() == type && c.getCapacity() > 0) {
                Measurement last = c.getLastMeasurement();
                if (last == null) {
                    disponivelTotal += c.getCapacity();
                } else {
                    double livre = c.getCapacity() - last.getValue();
                    if (livre > 0) {
                        disponivelTotal += livre;
                    }
                }
            }
        }
        return disponivelTotal;
    }

    public boolean isHighPriorityEligible(AidBox box, ItemType type, double minCapacity) {
        if (box == null || type == null) return false;
        PriorityLevel prio = box.getPriority();
        if (prio != PriorityLevel.URGENT && prio != PriorityLevel.HIGH) {
            return false;
        }

        double capDisponivel = getAvailableCapacityForItemType(box, type);
        return capDisponivel >= minCapacity;
    }

    @Override
    public DispatchOrder[] generateDispatchPlan(IInstitution inst, ItemType supplyType, double minAvailableCapacityNeeded) {
        if (inst == null || supplyType == null || inst.getAidBoxes() == null) {
            return new DispatchOrder[0];
        }

        AidBox[] boxes = inst.getAidBoxes();
        DispatchOrder[] temp = new DispatchOrder[boxes.length];
        int count = 0;

        for (int i = 0; i < boxes.length; i++) {
            AidBox box = boxes[i];
            if (box != null && isHighPriorityEligible(box, supplyType, minAvailableCapacityNeeded)) {
                double capDisponivel = getAvailableCapacityForItemType(box, supplyType);
                temp[count++] = new DispatchOrderImpl(box.getCode(), box.getPriority(), capDisponivel);
            }
        }

        DispatchOrder[] result = new DispatchOrder[count];
        for (int i = 0; i < count; i++) {
            result[i] = temp[i];
        }

        // Ordenação Manual Decrescente por Capacidade Disponível (Bubble Sort sem java.util)
        for (int i = 0; i < count - 1; i++) {
            for (int j = 0; j < count - 1 - i; j++) {
                if (result[j].getAvailableCapacity() < result[j + 1].getAvailableCapacity()) {
                    DispatchOrder tempOrder = result[j];
                    result[j] = result[j + 1];
                    result[j + 1] = tempOrder;
                }
            }
        }

        return result;
    }
}
