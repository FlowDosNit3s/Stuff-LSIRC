🚀 O Algoritmo Universal em 4 Passos (Perguntas 1a e 1b)

Passo 1: Atributos Privados e Construtores com Estados por Defeito (1a)
Declare todos os atributos como private. No construtor, inicialize os atributos recebidos e garanta que os estados enum por defeito (ex: ENABLED, ACTIVE, STANDBY) ou valores padrão de baterias/cargas são configurados corretamente.
Passo 2: Implementar Getters, Setters e Regras de Negócio (1a)
Respeite rigorosamente a assinatura dos métodos definidos nas interfaces. Em métodos que alterem estados (ex: setStatus ou descarregar bateria), aplique as regras condicionais exigidas no enunciado.
Passo 3: O Método equals(Object obj) Perfeito (1a)
Siga a estrutura padrão em 4 verificações:
1. if (this == obj) return true;
2. if (obj == null || getClass() != obj.getClass()) return false; (ou !(obj instanceof Tipo))
3. TipoImpl other = (TipoImpl) obj;
4. return this.code.equalsIgnoreCase(other.code); (ou comparação das chaves primárias indicadas).
Passo 4: Suite de Teste Completa no Método main() (1b)
Crie a classe de teste. No main, teste 100% das funcionalidades:
1. Instanciação de objetos válidos.
2. Invocação de todos os getters e exibição no ecrã com System.out.println.
3. Alteração de estados e verificação de regras de negócio.
4. Testes de equals com: (a) objetos iguais, (b) objetos com códigos diferentes, (c) comparação com null, e (d) comparação com objetos de outra classe.

public class RefrigeratedVehicleImpl implements RefrigeratedVehicle {

    private ItemType supplyType;
    private double maxCapacity;
    private double maxKilometers;
    private VehicleStatus status;

        this.code = code;
        this.supplyType = supplyType;
        this.maxCapacity = maxCapacity;
        this.maxKilometers = maxKilometers;
        this.status = VehicleStatus.ENABLED; // Estado por defeito: ENABLED
    }

    public String getCode() { return code; }

    @Override
    public ItemType getSupplyType() { return supplyType; }

    @Override
    public double getMaxCapacity() { return maxCapacity; }

    @Override
    public double getMaxKilometers() { return maxKilometers; }

    public VehicleStatus getStatus() { return status; }

    @Override
    public void setStatus(VehicleStatus status) {
        if (status != null) {
            this.status = status;
        }
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || !(obj instanceof Vehicle)) return false;
        Vehicle other = (Vehicle) obj;
        if (this.code == null || other.getCode() == null) return false;
        return this.code.equalsIgnoreCase(other.getCode());

    @Override
    public String toString() {
        return "RefrigeratedVehicle[Code=" + code + ", Type=" + supplyType + 
               ", Cap=" + maxCapacity + "kg, MaxKm=" + maxKilometers + "km, Status=" + status + "]";

public class SmartAidBoxImpl implements SmartAidBox {
    private String code;
    private String zone;
    private int maxContainers;

    private OperationalState state;

    public SmartAidBoxImpl(String code, String zone, int maxContainers) {
        this.code = code;
        this.zone = zone;
        this.maxContainers = maxContainers;
        this.batteryLevel = 100.0; // Bateria inicial 100%
        this.state = OperationalState.ACTIVE; // Estado por defeito: ACTIVE
    }

    @Override
    public String getCode() { return code; }

    @Override
    public String getZone() { return zone; }

    @Override
    public int getMaxContainers() { return maxContainers; }

    @Override
    public double getBatteryLevel() { return batteryLevel; }

    public void setState(OperationalState state) {
        if (state != null) {
            this.state = state;
        }
    }

    public void updateBattery(double newLevel) {
        this.batteryLevel = newLevel;
        if (this.batteryLevel <= 15.0) {
            this.state = OperationalState.MAINTENANCE;

    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || !(obj instanceof SmartAidBox)) return false;

        if (this.zone == null || other.getZone() == null) return false;

        return this.code.equalsIgnoreCase(other.getCode()) && this.zone.equalsIgnoreCase(other.getZone());
    }

    @Override
    public String toString() {
        return "SmartAidBox[Code=" + code + ", Zone=" + zone + ", MaxContainers=" + maxContainers + 
               ", Battery=" + batteryLevel + "%, State=" + state + "]";
    }

public class InsulatedContainerImpl implements InsulatedContainer {
    private String code;
    private ItemType type;
    private double capacity;
    private TempRating tempRating;
    private double currentTemp;
    private double maxTempLimit;

    public InsulatedContainerImpl(String code, ItemType type, double capacity, TempRating tempRating, double currentTemp, double maxTempLimit) {
        this.code = code;
        this.type = type;
        this.capacity = capacity;
        this.tempRating = tempRating;
        this.currentTemp = currentTemp;
        this.maxTempLimit = maxTempLimit;
    }

    @Override
    public String getCode() { return code; }

    @Override
    public ItemType getType() { return type; }

    @Override
    public double getCapacity() { return capacity; }

    @Override
    public TempRating getTempRating() { return tempRating; }

    @Override
    public double getCurrentTemp() { return currentTemp; }

    @Override
    public double getMaxTempLimit() { return maxTempLimit; }

    @Override
    public boolean isTemperatureCritical() {
        return currentTemp > maxTempLimit;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || !(obj instanceof InsulatedContainer)) return false;
        InsulatedContainer other = (InsulatedContainer) obj;

        if (this.code == null || other.getCode() == null) return false;

        return this.code.equalsIgnoreCase(other.getCode()) && this.tempRating == other.getTempRating();

    @Override
    public String toString() {
        return "InsulatedContainer[Code=" + code + ", Type=" + type + ", Rating=" + tempRating + 
               ", Temp=" + currentTemp + "ºC, Limit=" + maxTempLimit + "ºC, Critical=" + isTemperatureCritical() + "]";
    }
}

// ==================== PERGUNTA 1a ====================
public class AutonomousDroneVehicleImpl implements AutonomousDroneVehicle {
    private String code;
    private double maxCapacity;
    private double flightRangeKm;
    private double currentPayloadWeight;
    private DroneMode flightMode;

    public AutonomousDroneVehicleImpl(String code, double maxCapacity, double flightRangeKm) {
        this.code = code;
        this.maxCapacity = maxCapacity;
        this.flightRangeKm = flightRangeKm;
        this.currentPayloadWeight = 0.0;
        this.flightMode = DroneMode.STANDBY;
    }

    @Override
    public String getCode() { return code; }

    @Override
    public double getMaxCapacity() { return maxCapacity; }

    @Override
    public double getFlightRangeKm() { return flightRangeKm; }

    @Override
    public double getCurrentPayloadWeight() { return currentPayloadWeight; }

    @Override
    public DroneMode getFlightMode() { return flightMode; }

    @Override
    public boolean loadCargo(double weightKg) {
        if (flightMode != DroneMode.STANDBY || weightKg <= 0) return false;
        if (currentPayloadWeight + weightKg > maxCapacity) return false;

        currentPayloadWeight += weightKg;
        return true;
    }

    @Override
    public void triggerEmergency() {
        this.flightMode = DroneMode.EMERGENCY_LANDING;
        this.currentPayloadWeight = 0.0; // Descarte de emergência
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || !(obj instanceof AutonomousDroneVehicle)) return false;
        AutonomousDroneVehicle other = (AutonomousDroneVehicle) obj;

        if (this.code == null || other.getCode() == null) return false;

        boolean mesmoCodigo = this.code.equalsIgnoreCase(other.getCode());
        boolean mesmaAutonomia = Math.abs(this.flightRangeKm - other.getFlightRangeKm()) < 0.001;

        return mesmoCodigo && mesmaAutonomia;
    }

    @Override
    public String toString() {
        return "AutonomousDrone[Code=" + code + ", MaxCap=" + maxCapacity + "kg, Payload=" + currentPayloadWeight + 
               "kg, Range=" + flightRangeKm + "km, Mode=" + flightMode + "]";
    }
}

// ==================== PERGUNTA 1a ====================
public class HighPriorityAidBoxImpl implements HighPriorityAidBox {
    private String code;
    private String zone;
    private Container[] containers;
    private PriorityLevel priority;

    public HighPriorityAidBoxImpl(String code, String zone, Container[] containers, PriorityLevel priority) {
        this.code = code;
        this.zone = zone;
        this.containers = (containers != null) ? containers : new Container[0];
        this.priority = (priority != null) ? priority : PriorityLevel.LOW;
    }

    @Override
    public String getCode() { return code; }

    @Override
    public String getZone() { return zone; }

    @Override
    public Container[] getContainers() { return containers; }

    @Override
    public PriorityLevel getPriority() { return priority; }

    @Override
    public void setPriority(PriorityLevel priority) {
        if (priority != null) {
            this.priority = priority;
        }
    }

    @Override
    public int countContainers() {
        if (containers == null) return 0;
        int count = 0;
        for (int i = 0; i < containers.length; i++) {
            if (containers[i] != null) count++;
        }
        return count;
    }

    @Override
    public boolean isUrgentCollectionNeeded() {
        if (priority == PriorityLevel.URGENT) return true;
        return countContainers() > 5;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || !(obj instanceof HighPriorityAidBox)) return false;
        HighPriorityAidBox other = (HighPriorityAidBox) obj;

        if (this.code == null || other.getCode() == null) return false;
        if (this.zone == null || other.getZone() == null) return false;

        boolean mesmoCodigo = this.code.equalsIgnoreCase(other.getCode());
        boolean mesmaZona = this.zone.equalsIgnoreCase(other.getZone());
        boolean mesmaContagem = this.countContainers() == other.countContainers();

        return mesmoCodigo && mesmaZona && mesmaContagem;
    }

    @Override
    public String toString() {
        return "HighPriorityAidBox[Code=" + code + ", Zone=" + zone + ", Priority=" + priority + 
               ", ContainersValidos=" + countContainers() + ", Urgent=" + isUrgentCollectionNeeded() + "]";
    }
}
